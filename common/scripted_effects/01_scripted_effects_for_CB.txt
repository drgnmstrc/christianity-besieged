increase_lowest_monarch_power = {
	if = {
		limit = {
			AND = {
				$MONARCH_DIP$ = $MONARCH_ADM$
				$MONARCH_MIL$ = $MONARCH_ADM$
			}
		}
		change_adm = 1
	}
	else_if = {
		limit = {
			AND = {
				$MONARCH_ADM$ = $MONARCH_DIP$
				$MONARCH_MIL$ = $MONARCH_DIP$
			}
		}
		change_dip = 1
	}
	else_if = {
		limit = {
			AND = {
				$MONARCH_DIP$ = $MONARCH_MIL$
				$MONARCH_ADM$ = $MONARCH_MIL$
			}
		}
		change_mil = 1
	}
	else = {
		random_list = {
			33 = {
				change_adm = 1
			}
			33 = {
				change_dip = 1
			}
			33 = {
				change_mil = 1
			}
		}
	}
}

ashkharh_effect = {
	ashkharh_dynamic_name_change_effect = yes
	set_country_flag = has_ashkharh_country_name
	set_country_flag = has_ashkharh_color
	set_country_flag = has_overriden_name_flag
	set_country_flag = has_overriden_color_flag
	hidden_effect = {
		if = {
			limit = { religion = noreligion }
			change_religion = ROOT
		}
		if = {
			limit = { primary_culture = noculture }
			change_primary_culture = ROOT
		}
	}
	change_country_color = { color = { 73 67 102 } }
}

	set_country_flag = has_ashkharh_country_name
	set_country_flag = has_overriden_name_flag
	override_country_name = $tag$
}
