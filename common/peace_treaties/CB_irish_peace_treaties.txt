abdicate_high_kingship = {
	category = 6
	power_projection = humiliation_of_a_rival
	prestige_base = 1
	ae_base = 0

	warscore_cost = {
		# Which of the giver's provinces' war score costs are included and by how much. All entries default to 0. Any combination of entries can be used. Try negative values at your own risk.
		all_provinces = 0.0 # All the giver's provinces
		no_provinces = 20.0 # This is just a constant. Equivalent to a single province with with war score cost 1
		owner_keeps = 0.0 # The provinces that are not of any of the types below
		conquered = 1.0 # Provinces that are selected by a Demand Province treaty
		returned_core = 0.0 # Provinces that are selected by a Return Core treaty
		released_nation = 0.0 # Provinces that will become a part of a selected released-to-be nation
		cancelled_subject = 0.0 # This one should not be relevant here since Giver can't be a cancelled subject. Including anyway for completeness
		concede_colonial_region = 0.0 # Ditto
	}

	requires_is_allowed = yes
	
	effect = {
		if = {
			limit = { has_reform = irish_high_kingdom }
			remove_government_reform = irish_high_kingdom
			add_government_reform = irish_kingdom
		} else_if = {
			limit = { has_reform = irish_high_parlimentary_kingdom }
			remove_government_reform = irish_high_parlimentary_kingdom
			add_government_reform = irish_parlimentary_kingdom
		}
	}

	ai_weight = { # Defaults to always evaluate to 0
		export_to_variable = {
			variable_name = ai_value # Mandatory to have entry called "ai_value". Its value at the end of the ai_weight scope is what will be used
			value = 50
		}
		limit = { # if limit evaluates false, ai_weight will evaluate to 0 regardless
			owns_core_province = 1
		}
	}
}
