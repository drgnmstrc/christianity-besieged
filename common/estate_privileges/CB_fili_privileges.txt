## Allows `fili_divine_for_war_decision`
fili_war_diviners_privilege = {
	icon =  privilege_morale_damage

	loyalty = -0.12
	influence = 0.048

	on_granted = {
		enable_estate_action = {
			estate_action = FILI_WAR_DIVINATION
	    }
	}

	can_revoke = {
		NOT = {
			estate_influence = {
				estate = estate_fili
				influence = 30
			}
		}
	}

	ai_will_do = {
		factor = 0
	}
}
